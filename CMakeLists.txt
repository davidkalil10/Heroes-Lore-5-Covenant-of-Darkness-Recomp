cmake_minimum_required(VERSION 3.16)
project(hl5_recomp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W3 /O2)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-unused-function -O3)
endif()
find_package(SDL2 REQUIRED)
file(GLOB_RECURSE SRC_FILES src/*.cpp src/*.h)
add_executable(hl5 ${SRC_FILES})
target_include_directories(hl5 PRIVATE src ${SDL2_INCLUDE_DIRS})
if(TARGET SDL2::SDL2main AND WIN32)
  target_link_libraries(hl5 PRIVATE SDL2::SDL2main)
endif()
if(TARGET SDL2::SDL2)
  target_link_libraries(hl5 PRIVATE SDL2::SDL2)
else()
  target_link_libraries(hl5 PRIVATE ${SDL2_LIBRARIES})
endif()
if(MINGW)
  target_link_options(hl5 PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
endif()

# Native host tests; Android uses its own CMake project and Switch uses Makefile.switch.
include(CTest)
if(BUILD_TESTING)
  add_executable(hl5_display_tests tools/display_layout_test.cpp src/display_layout.cpp)
  target_include_directories(hl5_display_tests PRIVATE src ${SDL2_INCLUDE_DIRS})
  if(TARGET SDL2::SDL2)
    target_link_libraries(hl5_display_tests PRIVATE SDL2::SDL2)
  else()
    target_link_libraries(hl5_display_tests PRIVATE ${SDL2_LIBRARIES})
  endif()
  if(MINGW)
    target_link_options(hl5_display_tests PRIVATE -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME display_layout COMMAND hl5_display_tests)
endif()

find_package(Threads REQUIRED)
target_link_libraries(hl5 PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(hl5 PRIVATE winhttp)
elseif(UNIX)
  find_package(CURL REQUIRED)
  target_link_libraries(hl5 PRIVATE CURL::libcurl)
endif()

if(BUILD_TESTING)
  add_executable(hl5_service_tests tools/service_tests.cpp src/service_data.cpp src/services.cpp src/service_ui.cpp src/input_mapping.cpp src/display_layout.cpp)
  target_include_directories(hl5_service_tests PRIVATE src ${SDL2_INCLUDE_DIRS})
  if(TARGET SDL2::SDL2)
    target_link_libraries(hl5_service_tests PRIVATE SDL2::SDL2 Threads::Threads)
  else()
    target_link_libraries(hl5_service_tests PRIVATE ${SDL2_LIBRARIES} Threads::Threads)
  endif()
  if(MINGW)
    target_link_options(hl5_service_tests PRIVATE -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME services COMMAND hl5_service_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  set_tests_properties(services PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

if(BUILD_TESTING)
  target_sources(hl5_service_tests PRIVATE src/localization.cpp)
endif()
